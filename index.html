Database Setup Required

Your Supabase database needs to be set up. Please run the SQL script below in your Supabase SQL Editor:

  1. Go to supabase.com/dashboard
  2. Select your project
  3. Click "SQL Editor" in the left sidebar
  4. Paste the SQL below and click "Run"
-- Create users table CREATE TABLE IF NOT EXISTS public.users ( id SERIAL PRIMARY KEY, name TEXT NOT NULL, email TEXT NOT NULL UNIQUE, password TEXT NOT NULL, phone TEXT, age INTEGER, skills TEXT[] DEFAULT '{}', wants TEXT[] DEFAULT '{}', bio TEXT, credits INTEGER DEFAULT 25, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- Create messages table CREATE TABLE IF NOT EXISTS public.messages ( id SERIAL PRIMARY KEY, sender_id INTEGER NOT NULL, receiver_id INTEGER NOT NULL, text TEXT NOT NULL, read BOOLEAN DEFAULT false, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- Create reviews table CREATE TABLE IF NOT EXISTS public.reviews ( id SERIAL PRIMARY KEY, from_id INTEGER NOT NULL, to_id INTEGER NOT NULL, rating INTEGER NOT NULL, duration INTEGER NOT NULL, skill TEXT NOT NULL, text TEXT NOT NULL, from_name TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- Create bookings table CREATE TABLE IF NOT EXISTS public.bookings ( id SERIAL PRIMARY KEY, from_id INTEGER NOT NULL, to_id INTEGER NOT NULL, skill TEXT, date TEXT, time TEXT, duration INTEGER, message TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- Create notifications table CREATE TABLE IF NOT EXISTS public.notifications ( id SERIAL PRIMARY KEY, user_id INTEGER NOT NULL, text TEXT NOT NULL, icon TEXT, read BOOLEAN DEFAULT false, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- Enable RLS ALTER TABLE public.users ENABLE ROW LEVEL SECURITY; ALTER TABLE public.messages ENABLE ROW LEVEL SECURITY; ALTER TABLE public.reviews ENABLE ROW LEVEL SECURITY; ALTER TABLE public.bookings ENABLE ROW LEVEL SECURITY; ALTER TABLE public.notifications ENABLE ROW LEVEL SECURITY; -- Create permissive policies for testing (allow all) CREATE POLICY "Allow all" ON public.users FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "Allow all" ON public.messages FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "Allow all" ON public.reviews FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "Allow all" ON public.bookings FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "Allow all" ON public.notifications FOR ALL USING (true) WITH CHECK (true); -- Grant access GRANT ALL ON ALL TABLES IN SCHEMA public TO anon, authenticated; GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;

Frequently Asked Questions

Is SkillSwap free?

Yes! Everyone starts with 25 free credits. You earn more by teaching others and receiving good reviews. No money ever changes hands.

Who can join?

Anyone with a skill to share or a desire to learn. Whether you are a professional, student, hobbyist, or retiree.

How do sessions work?

Find someone with the skill you want, send them a message, and book a time. Sessions can be in-person or virtual.

Is it safe?

All members have public profiles with reviews. Start with short sessions, meet in public places, and trust your instincts.
